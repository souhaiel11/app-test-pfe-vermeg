package tn.pfe.demo;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertNotNull;
import static org.junit.jupiter.api.Assertions.assertTrue;

import org.apache.commons.lang3.StringUtils;
import org.apache.commons.text.StringEscapeUtils;
import org.junit.jupiter.api.DisplayName;
import org.junit.jupiter.api.Test;

class LegacyClientContractTest {

    /** Version lue dans le manifeste du jar réellement chargé — pas dans le pom. */
    private static String loadedVersion(Class<?> type) {
        return type.getPackage().getImplementationVersion();
    }

    @Test
    @DisplayName("NC-1 — le client historique exige commons-lang3 3.12.x exactement")
    void legacyClientRequiresLang3_3_12() {
        String version = loadedVersion(StringUtils.class);
        assertNotNull(version, "version de commons-lang3 illisible dans le manifeste");
        assertTrue(version.startsWith("3.12."),
            "Contrat du client historique rompu : commons-lang3 chargée en " + version
                + ", or le contrat impose 3.12.x");
    }

    @Test
    @DisplayName("NC-1 — le client historique exige commons-text 1.9 exactement")
    void legacyClientRequiresText_1_9() {
        assertEquals("1.9", loadedVersion(StringEscapeUtils.class),
            "Contrat du client historique rompu : commons-text n'est plus en 1.9");
    }
}
